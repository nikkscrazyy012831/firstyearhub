NAVIGATION LOGIC // Switches between different sections (Dashboard, Sem1, etc.) function show(id){ document.querySelectorAll('.page').forEach(p => p.classList.add('hidden')); document.getElementById(id).classList.remove('hidden'); } // 2. OWNER PANEL SECURITY const OWNER_PASSWORD = "admin"; // You can change this to your choice function unlock() { let enteredPw = document.getElementById("pw").value; if(enteredPw === OWNER_PASSWORD) { document.getElementById("unlockDiv").classList.add("hidden"); document.getElementById("uploadDiv").classList.remove("hidden"); } else { alert("Incorrect Admin Password"); } } // 3. UPLOAD TO CLOUD (Owner Action) function addAssignment() { let group = document.getElementById("groupSelect").value; let title = document.getElementById("fileTitle").value; let url = document.getElementById("fileLink").value; if(!title || !url) return alert("Please fill in all details!"); // Save data to Firebase Firestore db.collection("assignments").add({ group: group, title: title, url: url, timestamp: firebase.firestore.FieldValue.serverTimestamp() }) .then(() => { alert("Successfully posted to " + group + "!"); // Clear the inputs for next time document.getElementById("fileTitle").value = ""; document.getElementById("fileLink").value = ""; }) .catch(e => alert("Error saving: " + e.message)); } // 4. REAL-TIME SORTING (Student View) // This "Listens" to the cloud. If you add a file, it appears instantly! db.collection("assignments").orderBy("timestamp", "desc").onSnapshot((snapshot) => { const sem1Area = document.getElementById("sem1-list"); const sem2Area = document.getElementById("sem2-list"); // Clear old data sem1Area.innerHTML = ""; sem2Area.innerHTML = ""; snapshot.forEach((doc) => { let data = doc.data(); let cardHTML = `
${data.title}
Added: ${data.timestamp ? data.timestamp.toDate().toLocaleDateString() : 'Just now'}
`; // Check the 'group' tag and put it in the right list if (data.group === "sem1") { sem1Area.innerHTML += cardHTML; } else { sem2Area.innerHTML += cardHTML; } }); });
